cmake_minimum_required(VERSION 3.20)
project(FastBid VERSION 0.2.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "FastBid networking requires Linux; use WSL or a Linux container")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow)
option(ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(ENABLE_TSAN "Enable ThreadSanitizer (separate build)" OFF)
option(ENABLE_COVERAGE "Enable GCC coverage instrumentation" OFF)
if(ENABLE_COVERAGE)
    if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        message(FATAL_ERROR "Coverage configuration requires GCC")
    endif()
    if(ENABLE_SANITIZERS OR ENABLE_TSAN)
        message(FATAL_ERROR "Use a separate build for coverage and sanitizers")
    endif()
    add_compile_options(--coverage -O0 -g)
    add_link_options(--coverage)
endif()
if(ENABLE_SANITIZERS AND ENABLE_TSAN)
    message(FATAL_ERROR "Choose ASan/UBSan OR TSan")
endif()
if(ENABLE_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
elseif(ENABLE_TSAN)
    add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
    add_link_options(-fsanitize=thread)
endif()
find_package(Threads REQUIRED)
add_library(fastbid_core STATIC src/network/tcp_gateway.cpp src/network/udp_publisher.cpp src/platform/core_allocator.cpp)
target_include_directories(fastbid_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(fastbid_core PUBLIC Threads::Threads)
add_executable(fastbid src/main.cpp)
target_link_libraries(fastbid PRIVATE fastbid_core)
option(BUILD_TESTS "Build unit and socket integration tests" OFF)
if(BUILD_TESTS)
    enable_testing()
    include(FetchContent)
    FetchContent_Declare(googletest GIT_REPOSITORY https://github.com/google/googletest.git GIT_TAG f8d7d77c06936315286eb55f8de22cd23c188571)
    FetchContent_MakeAvailable(googletest)
    add_subdirectory(tests)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME socket_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:fastbid>)
    set_tests_properties(socket_integration PROPERTIES TIMEOUT 45)
    add_test(NAME runtime_contracts COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/runtime_contracts.py $<TARGET_FILE:fastbid>)
    set_tests_properties(runtime_contracts PROPERTIES TIMEOUT 90)
endif()
option(BUILD_BENCHMARKS "Build repeatable core microbenchmark and allocation guard" OFF)
if(BUILD_BENCHMARKS)
    add_executable(fastbid_bench benchmarks/core_benchmark.cpp)
    target_link_libraries(fastbid_bench PRIVATE fastbid_core)
endif()
